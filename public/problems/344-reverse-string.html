344. 反转字符串

观察 left、right 与字符数组在每一步如何变化

class Solution:    def reverseString(self, s: List[str]):        # 原地修改 s，不返回结果        left, right = 0, len(s) - 1        while left < right:            s[left], s[right] = s[right], s[left]            left += 1            right -= 1
